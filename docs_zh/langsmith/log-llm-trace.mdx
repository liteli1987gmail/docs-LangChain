当您直接调用 LLM 时（在 [LangChain](/oss/langchain/overview) 或 LangSmith [支持的集成](/langsmith/integrations) 之外），您需要提供特定的元数据，以便 LangSmith 可以显示令牌计数、计算成本，并让您使用正确的提供商和模型在 [Playground](/langsmith/prompt-engineering-concepts#playground) 中打开[运行](/langsmith/observability-concepts#runs)。

要获得功能完整的 LLM 追踪，有四个要求：

| 要求 | 怎么做 | 启用什么 |
|---|---|---|
| 1. 设置 [`run_type="llm"`](/langsmith/run-data-format#run-types) | 将 `run_type="llm"` 传递给 `@traceable` | LLM 特定渲染、令牌/成本显示 |
| 2. 格式化输入/输出 | 使用 OpenAI、Anthropic 或 LangChain 消息格式 | 结构化消息渲染、Playground 支持 |
| 3. 设置 `ls_provider` 和 `ls_model_name` | 在 `metadata` 中传递两者 | 成本追踪、Playground 模型选择 |
| 4. 提供令牌计数 | 在运行上设置 `usage_metadata` | 令牌计数和成本计算 |

<Note>
如果您使用的是 LangChain OSS、[OpenAI 包装器](/langsmith/trace-openai) 或 [Anthropic 包装器](/langsmith/trace-anthropic)，这些细节会自动处理。

本页面的示例使用 `traceable` 装饰器/包装器（适用于 Python 和 JS/TS 的推荐方法）。如果您直接使用 [RunTree](/langsmith/annotate-code#use-the-runtree-api) 或 [API](https://api.smith.langchain.com/redoc)，同样的要求也适用。
</Note>

## 消息格式

追踪自定义模型或自定义输入/输出格式时，它必须遵循 LangChain 格式、OpenAI completions 格式或 Anthropic 消息格式之一。有关更多详细信息，请参阅 [OpenAI Chat Completions](https://platform.openai.com/docs/api-reference/chat/create) 或 [Anthropic Messages](https://platform.claude.com/docs/en/api/messages) 文档。LangChain 格式是：

 <Expandable title="LangChain 格式">

<ParamField path="messages" type="array" required>
</ParamField>

## LangChain

如果您使用 LangChain，则无需担心手动设置这些字段。LangChain 会在内部处理所有必需的元数据。但是，如果您想直接设置这些值（例如，如果您使用自定义 LLM 类），以下是如何执行此操作的示例：

<CodeGroup>

```python Python
from langchain_openai import ChatOpenAI
from langsmith import traceable

llm = ChatOpenAI(model="gpt-4o-mini")

@traceable(
    project_name="openai-mini",
    run_type="llm",
    metadata={
        "ls_provider": "openai",
        "ls_model_name": "gpt-4o-mini",
    },
)
def call_openai(prompt: str) -> str:
    return llm.invoke(prompt)
```

```typescript TypeScript
import { ChatOpenAI } from "@langchain/openai";
import { traceable } from "langsmith/traceable";

const llm = new ChatOpenAI({
  model: "gpt-4o-mini",
});

const callOpenAI = traceable(
  async (prompt: string) => llm.invoke(prompt),
  {
    projectName: "openai-mini",
    runType: "llm",
    metadata: {
      ls_provider: "openai",
      ls_model_name: "gpt-4o-mini",
    },
  }
);
```

</CodeGroup>

## OpenAI

### Python

使用 `@traceable` 装饰器：

```python
from openai import OpenAI
from langsmith import traceable
from langsmith.utils import traceable

client = OpenAI()

@traceable(
    project_name="openai-5-mini",
    run_type="llm",
    metadata={
        "ls_provider": "openai",
        "ls_model_name": "gpt-4o-mini",
    },
)
def call_openai(prompt: str) -> str:
    response = client.chat.completions.create(
        model="gpt-4o-mini",
        messages=[
            {"role": "user", "content": prompt}
        ],
        # 重要：为每个响应包含 usage 以启用令牌计数
        # 这使 LangSmith 能够自动计算成本
        usage={
            "prompt_tokens": response.usage.prompt_tokens,
            "completion_tokens": response.usage.completion_tokens,
            "total_tokens": response.usage.total_tokens,
        },
    )
    return response.choices[0].message.content
```

使用 `Client.create_run` 的替代方法：

```python
import openai
from langsmith import Client

client = Client()

openai_client = openai.Client()

def predictoreau(prompt: str) -> str:
    with client.start_run(run_type="llm", project_name="openai-5-mini") as run:
        # 添加必需的元数据
        run.end_run(
            outputs={
                "raw": response.choices[0].message.content,
            },
            metadata={
                "ls_provider": "openai",
                "ls_model_name": "gpt-4o-mini",
            },
            # 添加 usage 以启用令牌计数
            extra={
                "usage": {
                    "prompt_tokens": response.usage.prompt_tokens,
                    "completion_tokens": response.usage.completion_tokens,
                    "total_tokens": response.usage.total_tokens,
                },
            },
        )
    return response.choices[0].message.content

response = openai_client.chat.completions.create(
    model="gpt-4o-mini",
    messages=[
        {"role": "user", "content": prompt}
    ],
)
predictoreau(prompt)
```

### TypeScript

使用 `@traceable` 装饰器：

```typescript
import OpenAI from "openai";
import { traceable } from "langsmith/traceable";

const client = new OpenAI();

const callOpenAI = traceable(
  async (prompt: string) => {
    const response = await client.chat.completions.create({
      model: "gpt-4o-mini",
      messages: [{ role: "user", content: prompt }],
    });
    return response.choices[0].message.content;
  },
  {
    projectName: "openai-5-mini",
    runType: "llm",
    metadata: {
      ls_provider: "openai",
      ls_model_name: "gpt-4o-mini",
    },
  }
);
```

使用 `Client.startRun` 的替代方法：

```typescript
import OpenAI from "openai";
import { Client } from "langsmith";

const client = new OpenAI();
const lsClient = new Client({ callerFinish: (run, _runTree) => {
    // 调用 finish 或 updateRun 以完成运行
    lsClient.finishRun(run.id, {
        outputs: { output: response.choices[0].message.content },
    });
}});

async function predictoreau(prompt: string) {
    const run = await lsClient.startRun({
        name: "call_openai",
        projectName: "openai-5-mini",
        runType: "llm",
        metadata: {
            ls_provider: "openai",
            ls_model_name: "gpt-4o-mini",
        },
    });

    const response = await client.chat.completions.create({
        model: "gpt-4o-mini",
        messages: [{ role: "user", content: prompt }],
    });

    await lsClient.updateRun(run.id, {
        outputs: { output: response.choices[0].message.content },
        extra: {
            usage: response.usage,
        },
    });
    return response.choices[0].message.content;
}
```

## Anthropic

### Python

```python
import anthropic
from langsmith import traceable

client = anthropic.Anthropic()

@traceable(
    project_name="claude-3-5",
    run_type="llm",
    metadata={
        "ls_provider": "anthropic",
        "ls_model_name": "claude-3-5-sonnet-20241022",
    },
)
def call_claude(prompt: str) -> str:
    response = client.messages.create(
        model="claude-3-5-sonnet-20241022",
        max_tokens=1024,
        messages=[
            {"role": "user", "content": prompt}
        ],
    )
    # 重要：为每个响应包含 usage 以启用令牌计数
    # 这使 LangSmith 能够自动计算成本
    return response.content[0].text
```

### TypeScript

```typescript
import Anthropic from "@anthropic-ai/sdk";
import { traceable } from "langsmith/traceable";

const client = new Anthropic();

const callClaude = traceable(
  async (prompt: string) => {
    const response = await client.messages.create({
      model: "claude-3-5-sonnet-20241022",
      max_tokens: 1024,
      messages: [{ role: "user", content: prompt }],
    });
    return response.content[0].text;
  },
  {
    projectName: "claude-3-5",
    runType: "llm",
    metadata: {
      ls_provider: "anthropic",
      ls_model_name: "claude-3-5-sonnet-20241022",
    },
  }
);
```
